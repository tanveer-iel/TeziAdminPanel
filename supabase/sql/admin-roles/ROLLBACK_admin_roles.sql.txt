-- Removes only what that script added. Leaves is_iel_admin() and all data tables untouched.
-- Policies on tables you do not own may need the owning role; "if exists" keeps it harmless.

do $$
begin
  begin drop policy if exists "fin_stmts_role_insert" on public.finanicial_statements;
        drop policy if exists "fin_stmts_role_update" on public.finanicial_statements;
        drop policy if exists "fin_stmts_role_delete" on public.finanicial_statements;

  begin drop policy if exists "fin_years_role_insert" on public.financial_years;
  exception when insufficient_privilege then raise notice 'not owner of financial_years: policy left'; end;
end
$$;
